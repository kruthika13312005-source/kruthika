<script>
    let currentCoords = "Not provided by customer";

    function navTo(pageId) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active-nav'));
        
        document.getElementById(pageId).classList.add('active');
        
        if(pageId === 'aboutPage') {
            document.getElementById('nav-about').classList.add('active-nav');
            loadDatabaseData(); 
        }
        if(pageId === 'servicePage') {
            document.getElementById('nav-services').classList.add('active-nav');
        }
        if(pageId === 'bookingPage') {
            document.getElementById('nav-booking').classList.add('active-nav');
            handleLocation();
        }
    }

    function handleLocation() {
        if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(
                (pos) => {
                    currentCoords = `http://googleusercontent.com/maps.google.com/6{pos.coords.latitude},${pos.coords.longitude}`;
                    const statusElement = document.getElementById('locStatus');
                    statusElement.innerText = "✅ Work-site GPS Coordinates Captured";
                    statusElement.style.background = "#d4edda";
                    statusElement.style.color = "#155724";
                },
                () => {
                    console.log("Location access declined.");
                }
            );
        }
    }

    window.onload = function() {
        loadDatabaseData();
    };

    function loadDatabaseData() {
        const localData = localStorage.getItem('kruthika_customer_bookings');
        const logs = localData ? JSON.parse(localData) : [];
        
        if(logs.length === 0) {
            document.getElementById('display').innerHTML = `<p style="font-size:13px; color:var(--text-muted); text-align:center;">No previous history found on this device.</p>`;
            return;
        }

        document.getElementById('display').innerHTML = logs.map(item => `
            <div class="db-item">
                <strong>${item.name}</strong> requested <strong>${item.service}</strong> <br>
                <small style="color: var(--text-muted);">${item.date}</small>
            </div>
        `).join('');
    }

    function submitBooking() {
        const name = document.getElementById('custName').value;
        const phone = document.getElementById('custPhone').value;
        const addr = document.getElementById('custAddr').value;
        const service = document.getElementById('serviceType').value;

        if(!name || !phone || !addr || !service) {
            alert("Please complete all requested booking information fields.");
            return;
        }

        const localData = localStorage.getItem('kruthika_customer_bookings');
        const logs = localData ? JSON.parse(localData) : [];
        const currentTimestamp = new Date().toLocaleString();
        
        logs.unshift({ name: name, service: service, phone: phone, address: addr, date: currentTimestamp });
        localStorage.setItem('kruthika_customer_bookings', JSON.stringify(logs));
        
        const officialWhatsAppNumber = "919980900314"; 
        
        const messageText = `*KRUTHIKA ELECTRICALS - NEW CLIENT REQUEST*%0A%0A` +
                            `*Customer Name:* ${encodeURIComponent(name)}%0A` +
                            `*Mobile Number:* ${encodeURIComponent(phone)}%0A` +
                            `*Required Service:* ${encodeURIComponent(service)}%0A` +
                            `*Site Location:* ${encodeURIComponent(addr)}%0A%0A` +
                            `*Google Maps Work-Site Link:*%0A${encodeURIComponent(currentCoords)}`;

        alert("Thank you! Your details have been securely logged. Click OK to instantly forward your booking to Kruthika Electricals on WhatsApp.");
        window.open(`https://wa.me/${officialWhatsAppNumber}?text=${messageText}`, '_blank');
        
        document.getElementById('custName').value = '';
        document.getElementById('custPhone').value = '';
        document.getElementById('custAddr').value = '';
        document.getElementById('serviceType').value = '';
        
        navTo('aboutPage');
    }
</script>

<script>
    // Booking logic
    function handleBooking() {
        alert("🎉 Booking successful! Thank you.");
    }

    // Star Rating logic
    const stars = document.querySelectorAll('.star');
    let selectedRating = 0;

    stars.forEach(star => {
        star.addEventListener('click', () => {
            selectedRating = star.getAttribute('data-value');
            updateStars(selectedRating);
        });

        star.addEventListener('mouseover', () => {
            updateStars(star.getAttribute('data-value'));
        });

        star.addEventListener('mouseleave', () => {
            updateStars(selectedRating);
        });
    });

    function updateStars(rating) {
        stars.forEach(star => {
            if (star.getAttribute('data-value') <= rating) {
                star.classList.add('active');
            } else {
                star.classList.remove('active');
            }
        });
    }

    // Feedback Submission logic
    function submitFeedback() {
        const reviewText = document.getElementById('reviewText').value;
        const suggestions = document.getElementById('suggestions').value;
        const otherOptions = document.getElementById('otherOptions').value;

        if (selectedRating === 0 && !reviewText && !suggestions && !otherOptions) {
            alert("Please fill out at least one field before submitting.");
            return;
        }

        // Displaying data (In real application, you'd send this to your backend server)
        console.log("Form Submitted:", {
            rating: selectedRating + " Stars",
            review: reviewText,
            suggestions: suggestions,
            otherOptions: otherOptions
        });

        alert("💌 Thank you for your feedback and suggestions!");
        
        // Clear fields after submit
        document.getElementById('reviewText').value = '';
        document.getElementById('suggestions').value = '';
        document.getElementById('otherOptions').value = '';
        selectedRating = 0;
        updateStars(0);
    }
</script>
